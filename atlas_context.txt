document.addEventListener("DOMContentLoaded", () => {
    console.log("[App] Инициализация высокопроизводительного интерфейса Atlas...");

    window.Store.init();
    window.HeaderView.init();
    window.CatalogView.init(); 
    window.ModalComponent.init();
    window.IconsView.init();
    window.AssignerView.init(); // Инициализация Помогатора

    const headerContainer = document.getElementById('app-header');
    const mainContainer = document.getElementById('app-main');

    let lastRouteHash = null;
    let lastRoutePath = null;
    let lastAdminState = null;
    let lastSortConfig = null;
    let lastSearchQuery = null;
    let lastItems = null; 
    
    const scrollCache = {};
    let isScrolling = false;
    let blockScrollCache = false; 

    window.addEventListener('popstate', () => {
        blockScrollCache = true;
    });

    document.addEventListener('click', (e) => {
        const link = e.target.closest('a');
        if (link && link.getAttribute('href') && link.getAttribute('href').startsWith('#')) {
            if (lastRouteHash && !document.body.classList.contains('no-scroll')) {
                scrollCache[lastRouteHash] = window.scrollY;
            }
            blockScrollCache = true; 
        }
        
        const backBtn = e.target.closest('#back-btn');
        if (backBtn) {
            e.preventDefault();
            blockScrollCache = true; 
            
            if (window.history.length > 2) {
                window.history.back();
            } else {
                window.location.hash = window.Store.getState().lastCatalogRoute || '#/';
            }
        }
    }, { capture: true }); 

    window.addEventListener('scroll', () => {
        if (!isScrolling && !blockScrollCache) {
            window.requestAnimationFrame(() => {
                if (lastRouteHash && !document.body.classList.contains('no-scroll') && !blockScrollCache) {
                    scrollCache[lastRouteHash] = window.scrollY;
                }
                isScrolling = false;
            });
            isScrolling = true;
        }
    }, { passive: true });

    window.Store.subscribe((state) => {
        const currentRouteHash = state.route.path + JSON.stringify(state.route.params);
        const currentRoutePath = state.route.path;
        const isModalOpen = !!state.route.params.modal;
        const currentSortConfig = JSON.stringify(state.sortConfig);
        const currentSearchQuery = state.searchQuery;

        if (isModalOpen) {
            if (!document.body.classList.contains('no-scroll')) {
                const scrollBarWidth = window.innerWidth - document.documentElement.clientWidth;
                document.body.style.paddingRight = `${scrollBarWidth}px`;
                document.body.classList.add('no-scroll');
            }
        } else {
            document.body.style.paddingRight = '';
            document.body.classList.remove('no-scroll');
        }

        const routeChanged = lastRouteHash !== currentRouteHash;
        const adminChanged = lastAdminState !== state.isAdmin;
        const sortChanged = lastSortConfig !== currentSortConfig;
        const searchChanged = lastSearchQuery !== currentSearchQuery;
        const itemsChanged = lastItems !== state.items; 

        if (routeChanged || adminChanged) {
            headerContainer.innerHTML = window.HeaderView.render(state);
        }

        if (routeChanged || adminChanged || sortChanged || searchChanged || itemsChanged) {
            
            let activeComponent = null;

            if (currentRoutePath === '/tierlist') {
                mainContainer.innerHTML = window.TierlistView.render(state);
                activeComponent = window.TierlistView;
            } else if (currentRoutePath === '/item') {
                mainContainer.innerHTML = window.ItemView.render(state);
                activeComponent = window.ItemView;
            } else if (currentRoutePath === '/icons') {
                mainContainer.innerHTML = window.IconsView.render(state);
                activeComponent = window.IconsView;
            } else if (currentRoutePath === '/assigner') {
                // Подключение рендера Помогатора
                mainContainer.innerHTML = window.AssignerView.render(state);
                activeComponent = window.AssignerView;
            } else {
                mainContainer.innerHTML = window.CatalogView.render(state);
                activeComponent = window.CatalogView;
            }

            requestAnimationFrame(() => {
                if (activeComponent && activeComponent.onMount) {
                    activeComponent.onMount();
                }

                setTimeout(() => {
                    if (!routeChanged && itemsChanged && !sortChanged && !searchChanged) {
                        // Оставляем скролл как есть
                    } else {
                        const scrollPosition = (!routeChanged && (sortChanged || searchChanged)) 
                            ? 0 
                            : (scrollCache[currentRouteHash] || 0);
                            
                        window.scrollTo({
                            top: scrollPosition,
                            left: 0,
                            behavior: 'auto'
                        });
                    }
                    
                    setTimeout(() => blockScrollCache = false, 50);
                    
                }, 10);
            });

            lastRouteHash = currentRouteHash;
            lastRoutePath = currentRoutePath;
            lastAdminState = state.isAdmin;
            lastSortConfig = currentSortConfig;
            lastSearchQuery = currentSearchQuery;
            lastItems = state.items;
        }
    });

    document.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyA') {
            e.preventDefault();
            const currentState = window.Store.getState().isAdmin;
            window.Store.setState({ isAdmin: !currentState });
        }
        
        if (e.key === 'Escape') {
            const state = window.Store.getState();
            if (state.route.params.modal) {
                window.ModalComponent.close();
            }
            const sortPopover = document.getElementById('sort-popover');
            if (sortPopover && sortPopover.classList.contains('active')) {
                sortPopover.classList.remove('active');
                document.getElementById('sort-toggle-btn').classList.remove('active');
            }
        }
    });

    window.Router.init();
});

window.Router = (() => {
    // Чтение URL (Deep Linking)
    const parseUrl = () => {
        const hash = window.location.hash.slice(1) || '/';
        const [path, queryString] = hash.split('?');
        
        const params = {};
        if (queryString) {
            const urlParams = new URLSearchParams(queryString);
            for (const [key, value] of urlParams) {
                params[key] = value;
            }
        }
        return { path: path || '/', params };
    };

    // Запись параметров в URL без мусора в истории (Replace) или с ним
    const setQueryParams = (newParams, replace = false) => {
        const current = parseUrl();
        const search = new URLSearchParams(current.params);
        
        for (const [key, val] of Object.entries(newParams)) {
            if (val === null || val === undefined || val === '') {
                search.delete(key);
            } else {
                search.set(key, val);
            }
        }
        
        const queryStr = search.toString();
        const newHash = queryStr ? `${current.path}?${queryStr}` : current.path;
        
        if (replace) {
            window.history.replaceState(null, '', `#${newHash}`);
            handleRouteChange(); // Триггерим вручную, т.к. replaceState не вызывает hashchange
        } else {
            window.location.hash = newHash;
        }
    };

    const handleRouteChange = () => {
        const route = parseUrl();
        console.log(`[Router] Навигация: ${route.path}`, route.params);
        
        const updates = { 
            route, 
            searchQuery: route.params.q ? route.params.q.toLowerCase() : '' 
        };

        // Запоминаем контекст каталога для Умной кнопки "Назад"
        // Игнорируем URL с открытыми окнами (modal), чтобы не вернуться в форму
        if (route.path === '/' || route.path === '/tierlist') {
            if (!route.params.modal) {
                updates.lastCatalogRoute = window.location.hash || '#/';
            }
        }

        window.Store.setState(updates);
    };

    const init = () => {
        window.addEventListener('hashchange', handleRouteChange);
        handleRouteChange(); // Первый рендер
    };

    return { init, parseUrl, setQueryParams };
})();

window.Store = (() => {
    let state = {
        items: [],
        isAdmin: false,
        route: { path: '/', params: {} },
        searchQuery: '',
        lastCatalogRoute: '#/', 
        // Добавлен дефолтный конфиг сортировки
        sortConfig: { by: 'title', dir: 'asc' }
    };

    const listeners = [];

    const init = () => {
        try {
            const localData = localStorage.getItem('atlas_data');
            if (localData) {
                state.items = JSON.parse(localData).items;
                console.log("[Store] Данные загружены из LocalStorage");
            } else if (window.__ATLAS_INITIAL_DATA__) {
                state.items = window.__ATLAS_INITIAL_DATA__.items;
                console.log("[Store] Данные загружены из манифеста");
            }
        } catch (error) {
            console.error("[Store] Ошибка загрузки данных:", error);
            state.items = [];
        }
    };

    const getState = () => state;

    const setState = (updates) => {
        state = { ...state, ...updates };
        
        if (updates.items) {
            localStorage.setItem('atlas_data', JSON.stringify({ items: state.items }));
        }

        listeners.forEach(listener => listener(state));
    };

    const subscribe = (listener) => {
        listeners.push(listener);
        listener(state); 
    };

    const exportData = () => {
        const dataStr = `window.__ATLAS_INITIAL_DATA__ = ${JSON.stringify({
            metadata: { lastUpdated: new Date().toISOString(), version: "1.5" },
            items: state.items
        }, null, 4)};`;
        
        const blob = new Blob([dataStr], { type: "text/javascript" });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = 'catalog.js';
        a.click();
        URL.revokeObjectURL(url);
    };

    return { init, getState, setState, subscribe, exportData };
})();

window.SearchComponent = (() => {
    let debounceTimer = null; // Таймер для предотвращения DOM Thrashing

    const render = (state) => {
        return `
            <div class="search-wrapper" id="search-wrapper">
                <div class="search-box">
                    ${window.Icons.search || '🔍'}
                    <input type="text" id="search-input" placeholder="Поиск по базе..." autocomplete="off">
                </div>
                <div class="search-dropdown" id="search-dropdown"></div>
            </div>
        `;
    };

    const init = () => {
        const header = document.getElementById('app-header');
        
        // Обработка ввода (Фильтрация и показ дропдауна)
        header.addEventListener('input', (e) => {
            if (e.target.id === 'search-input') {
                const val = e.target.value.toLowerCase().trim();
                const dropdown = document.getElementById('search-dropdown');
                
                // 1. МГНОВЕННОЕ ОБНОВЛЕНИЕ ДРОПДАУНА (Берем данные из текущего стейта, не перерисовывая каталог)
                if (!val) {
                    dropdown.innerHTML = '';
                    dropdown.classList.remove('active');
                } else {
                    const state = window.Store.getState();
                    const matches = state.items
                        .filter(i => i.title.toLowerCase().includes(val))
                        .slice(0, 5); // Топ 5 совпадений

                    if (matches.length > 0) {
                        dropdown.innerHTML = matches.map(item => `
                            <div class="search-dropdown-item" data-id="${item.id}">
                                <img src="${window.Helpers.escapeHtml(item.poster)}" class="search-dropdown-poster" onerror="this.style.display='none'">
                                <div class="search-dropdown-info">
                                    <span class="search-dropdown-title">${window.Helpers.escapeHtml(item.title)}</span>
                                    <span class="search-dropdown-type">${item.type}</span>
                                </div>
                            </div>
                        `).join('');
                    } else {
                        dropdown.innerHTML = `
                            <div class="search-dropdown-item">
                                <span style="color: var(--text-muted); font-size: 0.85rem;">Ничего не найдено</span>
                            </div>
                        `;
                    }
                    
                    dropdown.classList.add('active');
                }

                // 2. ОТЛОЖЕННОЕ ОБНОВЛЕНИЕ КАТАЛОГА (Debounce 300ms)
                // Сбрасываем старый таймер, если пользователь продолжает печатать
                clearTimeout(debounceTimer);
                
                debounceTimer = setTimeout(() => {
                    // Обновляем фильтр каталога через Store (вызовет тяжелый ререндер 1 раз)
                    window.Store.setState({ searchQuery: val });
                }, 300);
            }
        });

        // Обработка клика по элементу выпадающего списка
        header.addEventListener('click', (e) => {
            const searchItem = e.target.closest('.search-dropdown-item');
            if (searchItem && searchItem.dataset.id) {
                window.location.hash = `#/item?id=${searchItem.dataset.id}`;
                document.getElementById('search-input').value = '';
                document.getElementById('search-dropdown').classList.remove('active');
            }
        });

        // Закрытие списка при клике вне компонента
        document.addEventListener('click', (e) => {
            if (!e.target.closest('.search-wrapper')) {
                const dropdown = document.getElementById('search-dropdown');
                if (dropdown) dropdown.classList.remove('active');
            }
        });
    };

    return { render, init };
})();

                        </button>
                    </div>
                </div>
            </div>
        `;
    };

    const init = () => {
        const header = document.getElementById('app-header');

        header.addEventListener('click', (e) => {
            // Открытие/закрытие Popover'а
            const sortToggle = e.target.closest('#sort-toggle-btn');
            if (sortToggle) {
                sortToggle.classList.toggle('active');
                document.getElementById('sort-popover').classList.toggle('active');
                return;
            }

            // Выбор параметра сортировки (Алфавит, Оценка, Дата)
            const paramBtn = e.target.closest('.param-btn');
            if (paramBtn) {
                const newBy = paramBtn.dataset.by;
                const currentConfig = window.Store.getState().sortConfig;
                window.Store.setState({ sortConfig: { ...currentConfig, by: newBy } });
                
                document.querySelectorAll('.param-btn').forEach(btn => btn.classList.remove('active'));
                paramBtn.classList.add('active');
                return;
            }

            // Выбор направления сортировки
            const dirBtn = e.target.closest('.dir-btn');
            if (dirBtn) {
                const newDir = dirBtn.dataset.dir;
                const currentConfig = window.Store.getState().sortConfig;
                window.Store.setState({ sortConfig: { ...currentConfig, dir: newDir } });
                
                document.querySelectorAll('.dir-btn').forEach(btn => btn.classList.remove('active'));
                dirBtn.classList.add('active');
                return;
            }
        });

        // Закрытие поповера при клике вне его зоны
        document.addEventListener('click', (e) => {
            if (!e.target.closest('#sort-toggle-btn') && !e.target.closest('#sort-popover')) {
                const popover = document.getElementById('sort-popover');
                const toggle = document.getElementById('sort-toggle-btn');
                if (popover) popover.classList.remove('active');
                if (toggle) toggle.classList.remove('active');
            }
        });
    };

    return { render, init };
})();

window.Helpers = {
    escapeHtml: (unsafe) => {
        if (!unsafe) return '';
        return unsafe.toString().replace(/[&<"'>]/g, (match) => {
            const dict = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' };
            return dict[match];
        });
    },

    getTypeLabel: (type) => {
        const labels = {
            'games': 'Игры',
            'series': 'Аниме / Сериалы',
            'features': 'Полнометражки',
            'literature': 'Литература'
        };
        return labels[type] || 'Медиа';
    },

    getLifecycleStatus: (statuses) => {
        if (!statuses || !Array.isArray(statuses) || statuses.length === 0) return 'unassigned';
        if (statuses.includes('in_progress')) return 'in_progress';
        if (statuses.includes('endless')) return 'endless'; // Новое
        if (statuses.includes('completed')) return 'completed';
        if (statuses.includes('planned')) return 'planned';
        if (statuses.includes('dropped')) return 'dropped';
        return 'unassigned';
    },

    setLifecycleStatus: (statuses, newStatus) => {
        const lifecycleTypes = ['in_progress', 'endless', 'completed', 'planned', 'dropped'];
        const filtered = (statuses || []).filter(s => !lifecycleTypes.includes(s));
        if (newStatus) {
            filtered.push(newStatus);
        }
        return filtered;
    },

    getLifecycleDetails: (status) => {
        const dict = {
            'in_progress': { label: 'В процессе', icon: window.Icons.clock, colorClass: 'color-info' },
            'endless': { label: 'Сессионные', icon: window.Icons.infinity || '♾️', colorClass: 'color-warning' },

            'unassigned': { label: 'Без статуса', icon: window.Icons.grid, colorClass: 'color-primary' }
        };
        return dict[status] || dict['unassigned'];
    },

    // Конвертация ключей в визуальные иконки для карточки (Ось 2 и 3 + старые статусы прогресса)
    getStatusVisuals: (status, type) => {

            case 'played_self':
                let icon = window.Icons.tv;
                let title = 'Смотрел сам';
                if (type === 'games') { icon = window.Icons.gamepad; title = 'Играл сам'; }
                if (type === 'literature') { icon = window.Icons.book; title = 'Читал сам'; }
                return { svg: icon, colorClass: 'color-primary', title: title };
            
            case 'watched_youtube':
                return { svg: window.Icons.youtube, colorClass: 'color-youtube', title: 'Смотрел прохождение' };
            
            case 'listened_audio':
                return { svg: window.Icons.headphones || '🎧', colorClass: 'color-primary', title: 'Слушал аудиокнигу' };
            
            case 'watched_reaction':
                return { svg: window.Icons.users, colorClass: 'color-primary', title: 'Смотрел на реакции' };

            // ЭКСТРА
            case 'platinum':

window.AssignerView = (() => {
    let skippedIds = new Set();

    const render = (state) => {
        if (!state.isAdmin) {
            return `
                <div class="empty-state">
                    <div class="empty-state-icon">${window.Icons.lock}</div>
                    <h2>Доступ запрещен</h2>
                    <p>Помогатор доступен только администратору.</p>
                </div>
            `;
        }

        const currentType = state.route.params.type || 'all';

        const unassignedItems = state.items.filter(i => {
            const matchesType = currentType === 'all' || i.type === currentType;
            const noStatus = window.Helpers.getLifecycleStatus(i.statuses) === 'unassigned';
            return matchesType && noStatus && !skippedIds.has(i.id);
        });

        const tabsHtml = `
            <div style="margin-bottom: var(--spacing-xl); display: flex; flex-wrap: wrap; justify-content: center; gap: 10px;">
                <a href="#/assigner?type=all" class="btn ${currentType === 'all' ? 'btn-primary' : 'btn-secondary'}">Все подряд</a>
                <a href="#/assigner?type=games" class="btn ${currentType === 'games' ? 'btn-primary' : 'btn-secondary'}">Игры</a>
                <a href="#/assigner?type=features" class="btn ${currentType === 'features' ? 'btn-primary' : 'btn-secondary'}">Полнометражки</a>
                <a href="#/assigner?type=series" class="btn ${currentType === 'series' ? 'btn-primary' : 'btn-secondary'}">Аниме / Сериалы</a>
                <a href="#/assigner?type=literature" class="btn ${currentType === 'literature' ? 'btn-primary' : 'btn-secondary'}">Литература</a>
            </div>
        `;

        if (unassignedItems.length === 0) {
            const totalSkipped = state.items.filter(i => {
                const matchesType = currentType === 'all' || i.type === currentType;
                return matchesType && window.Helpers.getLifecycleStatus(i.statuses) === 'unassigned';
            }).length;

            let emptyMsg = `Вы распределили все произведения в этой категории! Отличная работа.`;
            let resetBtn = '';

                resetBtn = `<button class="btn btn-primary" id="assigner-reset-skips" style="margin-top: 20px;">Вернуться к пропущенным</button>`;
            }

            return tabsHtml + `
                <div class="empty-state">
                    <div class="empty-state-icon" style="color: var(--success-color);">${window.Icons.checkCircle}</div>
                    <h2>Всё готово</h2>
                    <p>${emptyMsg}</p>
                    ${resetBtn}
                </div>
            `;
        }

        const item = unassignedItems[0];
        const typeLabel = window.Helpers.getTypeLabel(item.type);

        return tabsHtml + `
            <div class="assigner-wrapper">

                <div class="assigner-card">

                    </div>
                </div>

                    <button class="assigner-btn in_progress" data-status="in_progress" title="В процессе">

                    <button class="assigner-btn endless" data-status="endless" title="Сессионка">

                        ${window.Icons.arrowDown}
                    </button>
                </div>
            </div>
        `;
    };

    const init = () => {
        const main = document.getElementById('app-main');
        
        main.addEventListener('click', (e) => {
            if (!window.Store.getState().route.path.startsWith('/assigner')) return;

            if (e.target.closest('#assigner-reset-skips')) {
                skippedIds.clear();
                main.innerHTML = window.AssignerView.render(window.Store.getState());
                return;
            }

            const skipBtn = e.target.closest('#assigner-skip-btn');
            if (skipBtn) {
                const id = skipBtn.closest('.assigner-controls').dataset.id;
                skippedIds.add(id);
                main.innerHTML = window.AssignerView.render(window.Store.getState());
                return;
            }

            const statusBtn = e.target.closest('.assigner-btn[data-status]');
            if (statusBtn) {
                const id = statusBtn.closest('.assigner-controls').dataset.id;
                const newStatus = statusBtn.dataset.status;
                
                const currentItems = window.Store.getState().items;
                const updatedItems = currentItems.map(i => {
                    if (i.id === id) {
                        return { 
                            ...i, 
                            statuses: window.Helpers.setLifecycleStatus(i.statuses, newStatus) 
                        };
                    }
                    return i;
                });
                
                window.Store.setState({ items: updatedItems });
                return;
            }
        });
    };

    return { render, init };
})();

                'dropped': {},

                    groupedByStatus[lifecycle][c].push(item);
                });
            });

            // Строгий порядок глобальных секций (с endless)
            const statusOrder = ['in_progress', 'endless', 'completed', 'planned', 'dropped', 'unassigned'];

                const categories = Object.keys(collectionsObj);
                
                if (categories.length === 0) return; 

                const statusDetails = window.Helpers.getLifecycleDetails(statusKey);
                
                const uniqueItems = new Set();
                categories.forEach(cat => collectionsObj[cat].forEach(i => uniqueItems.add(i.id)));
                
                html += `
                    <div class="global-status-section" data-status="${statusKey}">
                        <div class="global-status-header ${statusDetails.colorClass}">
                            <div class="global-status-icon">${statusDetails.icon}</div>
                            <h2>${statusDetails.label} <span>${uniqueItems.size}</span></h2>
                        </div>
                `;

                const sortedCategories = categories.sort((a, b) => {
                    if (a === 'Без категории') return 1;
                    if (b === 'Без категории') return -1;
                    return a.localeCompare(b);
                });

                sortedCategories.forEach(category => {
                    collectionsObj[category].sort((a, b) => {
                        let valA = (a.title || '').toLowerCase();
                        let valB = (b.title || '').toLowerCase();
                        let compareResult = valA.localeCompare(valB);
                        if (sortConfig.dir === 'desc') compareResult *= -1;
                        return compareResult;
                    });

                    html += `
                        <section class="collection-group">
                            <h3 class="collection-title">
                                ${window.Helpers.escapeHtml(category)} <span>${collectionsObj[category].length}</span>
                            </h3>
                            <div class="grid">
                                ${collectionsObj[category].map(item => window.CardComponent.render(item, isAdmin)).join('')}
                            </div>
                        </section>
                    `;
                });

                html += `</div>`; 
            });

            return html;
        } 
        else {
            filteredItems.sort((a, b) => {
                let valA, valB;
                
                if (sortConfig.by === 'date') {
                    const dateA = a.completedAt ? new Date(a.completedAt).getTime() : 0;
                    const dateB = b.completedAt ? new Date(b.completedAt).getTime() : 0;
                    valA = isNaN(dateA) ? 0 : dateA;
                    valB = isNaN(dateB) ? 0 : dateB;
                } else if (sortConfig.by === 'score') {
                    valA = a.score || 0;
                    valB = b.score || 0;
                }

                let compareResult = 0;
                if (valA < valB) compareResult = -1;
                else if (valA > valB) compareResult = 1;
                
                if (sortConfig.dir === 'desc') compareResult *= -1;

                if (compareResult === 0) {
                    return (a.title || '').localeCompare(b.title || '');
                }
                
                return compareResult;
            });

            return `
                <section class="collection-group flat-grid">
                    <div class="grid">
                        ${filteredItems.map(item => window.CardComponent.render(item, isAdmin)).join('')}
                    </div>
                </section>
            `;
        }
    };

    const onMount = () => {
        const cards = document.querySelectorAll('.card-hidden');
        if (cards.length === 0) return;

        let delayIndex = 0;
        let resetTimer = null;

        const observer = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    const card = entry.target;
                    card.style.animationDelay = `${delayIndex * 40}ms`;
                    card.classList.remove('card-hidden');
                    card.classList.add('card-visible');
                    
                    observer.unobserve(card);
                    delayIndex++;

                    clearTimeout(resetTimer);
                    resetTimer = setTimeout(() => { delayIndex = 0; }, 100);
                }
            });
        }, { rootMargin: '100px', threshold: 0.1 });

        cards.forEach(card => observer.observe(card));
    };

    const init = () => {
        const main = document.getElementById('app-main');
        
        main.addEventListener('click', (e) => {
            const deleteBtn = e.target.closest('.delete-btn');
            if (deleteBtn) {
                e.preventDefault(); 
                e.stopPropagation();
                
                const id = deleteBtn.dataset.id;
                if (confirm('Вы уверены, что хотите удалить эту запись?')) {
                    const newItems = window.Store.getState().items.filter(i => i.id !== id);
                    window.Store.setState({ items: newItems });
                    if (window.Router.parseUrl().params.id === id) {
                        window.location.hash = '#/';
                    }
                }
                return;
            }

            const editBtn = e.target.closest('.edit-btn');
            if (editBtn) {
                e.preventDefault(); 
                e.stopPropagation();
                
                const id = editBtn.dataset.id;
                window.Router.setQueryParams({ modal: 'edit', id: id });
                return;
            }
        });
    };

    return { render, init, onMount };
})();

window.TierlistView = (() => {
    const tiers = ['S', 'A', 'B', 'C', 'D'];

    const render = (state) => {
        const { items, route, searchQuery, isAdmin } = state;
        const filterType = route.params.filter || 'all'; 
        
        let targetItems = items.filter(i => {
            const matchesType = filterType === 'all' || i.type === filterType;
            const matchesSearch = searchQuery ? i.title.toLowerCase().includes(searchQuery) : true;
            return matchesType && matchesSearch;
        });

        const grouped = { S: [], A: [], B: [], C: [], D: [], unassigned: [] };
        
        targetItems.forEach(item => {
            if (item.tier && tiers.includes(item.tier)) {
                grouped[item.tier].push(item);
            } else {
                grouped.unassigned.push(item);
            }
        });

        return `
            <div style="margin-bottom: var(--spacing-lg); display: flex; flex-wrap: wrap; gap: 10px;">
                <a href="#/tierlist?filter=all" class="btn ${filterType === 'all' ? 'btn-primary' : 'btn-secondary'}">Все</a>
                <a href="#/tierlist?filter=games" class="btn ${filterType === 'games' ? 'btn-primary' : 'btn-secondary'}">Игры</a>
                <a href="#/tierlist?filter=features" class="btn ${filterType === 'features' ? 'btn-primary' : 'btn-secondary'}">Полнометражки</a>
                <a href="#/tierlist?filter=series" class="btn ${filterType === 'series' ? 'btn-primary' : 'btn-secondary'}">Аниме/Сериалы</a>
                <a href="#/tierlist?filter=literature" class="btn ${filterType === 'literature' ? 'btn-primary' : 'btn-secondary'}">Литература</a>
            </div>

            <div class="tier-board">
                ${tiers.map(tier => `
                    <div class="tier-row" data-tier="${tier}">
                        <div class="tier-label tier-${tier}">${tier}</div>
                        <div class="tier-items drop-zone">
                            ${grouped[tier].map(item => `
                                <a href="#/item?id=${item.id}" class="tier-item-card" data-id="${item.id}" draggable="${isAdmin}">
                                    <img src="${window.Helpers.escapeHtml(item.poster)}" alt="${window.Helpers.escapeHtml(item.title)}" onerror="this.style.display='none'">
                                    <div class="tier-item-tooltip">${window.Helpers.escapeHtml(item.title)}</div>
                                </a>
                            `).join('')}
                        </div>
                    </div>
                `).join('')}
            </div>

            ${isAdmin ? `
                <div class="tier-pool drop-zone" data-tier="unassigned">
                    <div class="tier-pool-title">
                        ${window.Icons.grid || ''} Нераспределенные произведения (${grouped.unassigned.length})
                    </div>
                    <div class="tier-items" style="min-height: 80px;">
                        ${grouped.unassigned.map(item => `
                            <a href="#/item?id=${item.id}" class="tier-item-card" data-id="${item.id}" draggable="true">
                                <img src="${window.Helpers.escapeHtml(item.poster)}" alt="${window.Helpers.escapeHtml(item.title)}" onerror="this.style.display='none'">
                                <div class="tier-item-tooltip">${window.Helpers.escapeHtml(item.title)}</div>
                            </a>
                        `).join('')}
                    </div>
                </div>
            ` : ''}
        `;
    };

    const onMount = () => {
        const state = window.Store.getState();
        if (!state.isAdmin) return;

        const draggables = document.querySelectorAll('.tier-item-card');
        const dropZones = document.querySelectorAll('.drop-zone');
        
        let draggedItem = null;
        let placeholder = document.createElement('div');
        placeholder.classList.add('tier-placeholder');

        // Умная 2D логика вычисления позиции, не мешающая скроллу
        const getDragAfterElement = (container, x, y) => {
            const draggableElements = [...container.querySelectorAll('.tier-item-card:not(.dragging)')];

            return draggableElements.find(child => {
                const box = child.getBoundingClientRect();
                const isSameRow = y >= box.top && y <= box.bottom;
                
                if (isSameRow) {
                    return x < box.left + box.width / 2;
                }
                if (y < box.top) return true;
                
                return false;
            });
        };

        draggables.forEach(draggable => {
            draggable.addEventListener('dragstart', (e) => {
                e.dataTransfer.setData('text/plain', draggable.dataset.id);
                e.dataTransfer.effectAllowed = 'move';
                
                draggedItem = draggable;
                setTimeout(() => draggable.classList.add('dragging'), 0);
            });

            draggable.addEventListener('dragend', () => {
                draggable.classList.remove('dragging');
                if (placeholder.parentNode) {
                    placeholder.parentNode.removeChild(placeholder);
                }
                
                document.querySelectorAll('.drag-over').forEach(zone => {
                    zone.classList.remove('drag-over');
                });
                draggedItem = null;
            });
            
            draggable.addEventListener('click', (e) => {
                if (draggable.classList.contains('dragging')) e.preventDefault();
            });
        });

        dropZones.forEach(zone => {
            zone.addEventListener('dragover', (e) => {
                e.preventDefault(); 
                
                const row = zone.closest('.tier-row') || zone.closest('.tier-pool');
                if (row) row.classList.add('drag-over');

                const itemsContainer = zone.classList.contains('tier-items') ? zone : zone.querySelector('.tier-items');
                const afterElement = getDragAfterElement(itemsContainer, e.clientX, e.clientY);
                
                // Избегаем бесконечных перерисовок DOM, двигаем фантом только если позиция реально изменилась
                if (placeholder.parentNode !== itemsContainer || placeholder.nextElementSibling !== afterElement) {
                    if (afterElement == null) {
                        itemsContainer.appendChild(placeholder);
                    } else {
                        itemsContainer.insertBefore(placeholder, afterElement);
                    }
                }
            });

            zone.addEventListener('dragleave', (e) => {
                const row = zone.closest('.tier-row') || zone.closest('.tier-pool');
                if (row && !row.contains(e.relatedTarget)) {
                    row.classList.remove('drag-over');
                }
            });

            zone.addEventListener('drop', (e) => {
                e.preventDefault();
                if (!draggedItem) return;

                const itemId = draggedItem.dataset.id;
                const targetZone = zone.closest('[data-tier]');
                if (!targetZone) return;

                let newTier = targetZone.dataset.tier;
                if (newTier === 'unassigned') newTier = ''; 

                // 1. Ищем, ПЕРЕД какой карточкой мы сбросили элемент
                let nextNode = placeholder.nextElementSibling;
                // Игнорируем саму себя, если сбросили рядом с исходным местом
                while (nextNode && nextNode.classList.contains('dragging')) {
                    nextNode = nextNode.nextElementSibling;
                }
                const nextItemId = nextNode ? nextNode.dataset.id : null;

                const currentItems = [...window.Store.getState().items];
                
                // 2. ВЫРЕЗАЕМ карточку со старого места
                const draggedIdx = currentItems.findIndex(i => i.id === itemId);
                if (draggedIdx === -1) return;
                
                const [draggedObj] = currentItems.splice(draggedIdx, 1);
                draggedObj.tier = newTier;

                // 3. ВСТАВЛЯЕМ на новое место (настоящий swap в массиве базы данных)
                if (nextItemId) {
                    const nextIdx = currentItems.findIndex(i => i.id === nextItemId);
                    if (nextIdx !== -1) {
                        currentItems.splice(nextIdx, 0, draggedObj);
                    } else {
                        currentItems.push(draggedObj);
                    }
                } else {
                    currentItems.push(draggedObj);
                }

                // Сохраняем и бесшовно обновляем экран
                window.Store.setState({ items: currentItems });
            });
        });
    };

    return { render, onMount };
})();
