document.addEventListener("DOMContentLoaded", () => {
    console.log("Atlas | Инициализация интерфейса...");

    // 1. Инициализируем хранилище и события модулей
    window.Store.init();
    window.HeaderView.init();
    window.CatalogView.init();
    window.ModalView.init(); // Добавили инициализацию модалки

    const headerContainer = document.getElementById('app-header');
    const mainContainer = document.getElementById('app-main');

    let lastRouteHash = null;
    let lastAdminState = null;

    // 2. Подписываемся на изменения Store
    window.Store.subscribe((state) => {
        const currentRouteHash = state.route.path + JSON.stringify(state.route.params);
        
        // Перерисовываем Header только если изменился путь или режим админа
        if (lastRouteHash !== currentRouteHash || lastAdminState !== state.isAdmin) {
            headerContainer.innerHTML = window.HeaderView.render(state);
            lastRouteHash = currentRouteHash;
            lastAdminState = state.isAdmin;
        }

        // Каталог перерисовываем при любом изменении
        mainContainer.innerHTML = window.CatalogView.render(state);
    });

    // 3. Глобальный шорткат для входа в режим Админа
    document.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.code === 'KeyA') {
            e.preventDefault();
            const currentState = window.Store.getState().isAdmin;
            window.Store.setState({ isAdmin: !currentState });
        }
    });

    // 4. Запуск роутера
    window.Router.init();
});

window.Router = (() => {
    const parseUrl = () => {
        const hash = window.location.hash.slice(1) || '/';
        const [path, queryString] = hash.split('?');
        
        const params = {};
        if (queryString) {
            const urlParams = new URLSearchParams(queryString);
            for (const [key, value] of urlParams) {
                params[key] = value;
            }
        }
        return { path, params };
    };

    const handleRouteChange = () => {
        const route = parseUrl();
        window.Store.setState({ route });
    };

    const init = () => {
        window.addEventListener('hashchange', handleRouteChange);
        // Триггерим первый раз при загрузке
        handleRouteChange();
    };

    return { init };
})();

window.Store = (() => {
    let state = {
        items: [],
        isAdmin: false,
        route: { path: '/', params: {} },
        searchQuery: ''
    };

    const listeners = [];

    // Инициализация данных: приоритет у LocalStorage, если пусто - берем из файла
    const init = () => {
        try {
            const localData = localStorage.getItem('atlas_data');
            if (localData) {
                state.items = JSON.parse(localData).items;
            } else if (window.__ATLAS_INITIAL_DATA__) {
                state.items = window.__ATLAS_INITIAL_DATA__.items;
            }
        } catch (error) {
            console.error("Ошибка загрузки данных в Store:", error);
            state.items = [];
        }
    };

    const getState = () => state;

    const setState = (updates) => {
        state = { ...state, ...updates };
        
        // Если обновились items, сохраняем в localStorage
        if (updates.items) {
            localStorage.setItem('atlas_data', JSON.stringify({ items: state.items }));
        }

        // Оповещаем подписчиков
        listeners.forEach(listener => listener(state));
    };

    const subscribe = (listener) => {
        listeners.push(listener);
        // Сразу вызываем для первоначальной отрисовки
        listener(state);
    };

    // Экспорт базы в файл
    const exportData = () => {
        const dataStr = `window.__ATLAS_INITIAL_DATA__ = ${JSON.stringify({
            metadata: { lastUpdated: new Date().toISOString(), version: "1.0" },
            items: state.items
        }, null, 4)};`;
        
        const blob = new Blob([dataStr], { type: "text/javascript" });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = 'catalog.js';
        a.click();
        URL.revokeObjectURL(url);
    };

    return { init, getState, setState, subscribe, exportData };
})();

window.CatalogView = (() => {
    const statusLabels = {
        'played_self': 'Играл(а) / Читал(а) сам(а)',
        'watched_youtube': 'Смотрел(а) на YouTube',
        'watched_reaction': 'Смотрел(а) реакцию',
        'completed': 'Пройдено/Просмотрено',
        'in_progress': 'В процессе',
        'dropped': 'Брошено',
        'planned': 'В планах'
    };

    const formatDate = (dateStr) => {
        if (!dateStr) return '';
        const d = new Date(dateStr);
        return d.toLocaleDateString('ru-RU');
    };

    const renderCard = (item, isAdmin) => {
        let progressText = '';
        if (item.meta && (item.meta.currentProgress || item.meta.totalProgress)) {
            const current = item.meta.currentProgress || '?';
            const total = item.meta.totalProgress || '?';
            const label = item.type === 'literature' ? 'Главы/Стр:' : 'Серии:';
            progressText = `<span>📊 ${label} ${current} / ${total}</span>`;
        }

        const dateText = item.completedAt ? `<span>📅 Пройдено: ${formatDate(item.completedAt)}</span>` : '';

        return `
        <div class="card">
            <div class="card-poster-wrapper">
                <img src="${item.poster}" alt="${item.title}" class="card-poster" onerror="this.style.display='none'">
                <div class="card-score tier-${item.tier}">
                    ${item.tier ? item.tier : (item.score ? item.score + '/10' : '-')}
                </div>
                ${isAdmin ? `
                    <div class="card-admin-actions">
                        <button class="btn-icon edit-btn" data-id="${item.id}">${window.Icons.edit}</button>
                        <button class="btn-icon danger delete-btn" data-id="${item.id}">${window.Icons.trash}</button>
                    </div>
                ` : ''}
            </div>
            <div class="card-info">
                <h3 class="card-title" title="${item.title}">${item.title}</h3>
                
                ${(dateText || progressText) ? `
                    <div class="card-meta-info">
                        ${progressText}
                        ${dateText}
                    </div>
                ` : ''}

                <div class="card-statuses">
                    ${(item.statuses || []).map(st => `<span class="status-badge">${statusLabels[st] || st}</span>`).join('')}
                </div>
            </div>
        </div>
        `;
    };

    const render = (state) => {
        const { items, route, searchQuery, isAdmin } = state;
        const currentType = route.params.type || 'all';
        
        let filteredItems = items.filter(item => {
            const matchType = currentType === 'all' || item.type === currentType;
            const matchSearch = item.title.toLowerCase().includes(searchQuery);
            return matchType && matchSearch;
        });

        if (route.path === '/tierlist') {
            return `<div class="empty-state">
                        <div class="empty-state-icon">${window.Icons.emptyBox}</div>
                        <h2>Раздел в разработке</h2>
                        <p>Тир-листы появятся в следующем обновлении.</p>
                    </div>`;
        }

        // Красивое состояние "Ничего не найдено"
        if (filteredItems.length === 0) {
            return `
                <div class="empty-state">
                    <div class="empty-state-icon">${window.Icons.emptyBox}</div>
                    <h2>Ничего не найдено</h2>
                    <p>Здесь пока пусто. Попробуйте изменить поисковый запрос или добавьте новую карточку через панель администратора.</p>
                </div>
            `;
        }

        const grouped = {};
        filteredItems.forEach(item => {
            const colls = (item.collections && item.collections.length > 0) ? item.collections : ['Без категории'];
            colls.forEach(c => {
                if (!grouped[c]) grouped[c] = [];
                grouped[c].push(item);
            });
        });

        const sortedCategories = Object.keys(grouped).sort((a, b) => {
            if (a === 'Без категории') return 1;
            if (b === 'Без категории') return -1;
            return a.localeCompare(b);
        });

        return sortedCategories.map(category => `
            <div class="collection-group">
                <h2 class="collection-title">
                    ${category} <span>${grouped[category].length}</span>
                </h2>
                <div class="grid">
                    ${grouped[category].map(item => renderCard(item, isAdmin)).join('')}
                </div>
            </div>
        `).join('');
    };

    const init = () => {
        const main = document.getElementById('app-main');
        
        main.addEventListener('click', (e) => {
            const deleteBtn = e.target.closest('.delete-btn');
            if (deleteBtn) {
                const id = deleteBtn.dataset.id;
                if (confirm('Вы уверены, что хотите удалить эту запись?')) {
                    const newItems = window.Store.getState().items.filter(i => i.id !== id);
                    window.Store.setState({ items: newItems });
                }
            }

            const editBtn = e.target.closest('.edit-btn');
            if (editBtn) {
                const id = editBtn.dataset.id;
                const itemToEdit = window.Store.getState().items.find(i => i.id === id);
                if (itemToEdit) {
                    window.ModalView.open(itemToEdit);
                }
            }
        });
    };

    return { render, init };
})();

                        </div>
                        
                        <div class="modal-footer">
                            <button type="button" class="btn btn-secondary" id="cancel-modal">Отмена</button>
                            <button type="submit" class="btn btn-primary">Сохранить</button>
                        </div>
                    </form>
                </div>
            </div>
        `;
    };

    const open = (item = null) => {
        const modalContainer = document.getElementById('app-modal');
        modalContainer.innerHTML = render(item);
        
        // Добавляем слушатель для переключения видимости полей прогресса
        const typeSelect = document.getElementById('type-select');
        const progressFields = document.getElementById('progress-fields');
        if (typeSelect && progressFields) {
            typeSelect.addEventListener('change', (e) => {
                if (e.target.value === 'series' || e.target.value === 'literature') {
                    progressFields.style.display = 'grid';
                } else {
                    progressFields.style.display = 'none';
                    // Очищаем значения при скрытии
                    document.querySelector('input[name="currentProgress"]').value = '';
                    document.querySelector('input[name="totalProgress"]').value = '';
                }
            });
        }
    };

    const close = () => {
        const modalContainer = document.getElementById('app-modal');
        modalContainer.innerHTML = '';
        currentItem = null;
    };

    const init = () => {
        const modalContainer = document.getElementById('app-modal');

        modalContainer.addEventListener('click', (e) => {
            if (e.target.closest('#close-modal') || 
                e.target.closest('#cancel-modal') || 
                e.target.id === 'modal-overlay') {
                close();
            }
        });

        modalContainer.addEventListener('submit', (e) => {
            if (e.target.id === 'atlas-form') {
                e.preventDefault();
                const formData = new FormData(e.target);
                
                const selectedStatuses = formData.getAll('statuses');
                
                const collectionsStr = formData.get('collections');
                const parsedCollections = collectionsStr
                    .split(',')
                    .map(c => c.trim())
                    .filter(c => c.length > 0);

                // Собираем мета-данные
                const currentP = formData.get('currentProgress');
                const totalP = formData.get('totalProgress');
                let metaData = {};
                if (currentP || totalP) {
                    metaData = {
                        currentProgress: currentP ? Number(currentP) : null,
                        totalProgress: totalP ? Number(totalP) : null
                    };
                }

                const newItem = {
                    id: currentItem?.id || Date.now().toString(),
                    title: formData.get('title'),
                    type: formData.get('type'),
                    collections: parsedCollections,
                    statuses: selectedStatuses,
                    score: Number(formData.get('score')) || null,
                    tier: formData.get('tier') || null,
                    poster: formData.get('poster'),
                    completedAt: formData.get('completedAt') || null,
                    review: formData.get('review')
                };

                // Добавляем meta только если там есть данные
                if (Object.keys(metaData).length > 0) {
                    newItem.meta = metaData;
                }

                const currentItems = window.Store.getState().items;
                let newItemsList;

                if (currentItem?.id) {
                    newItemsList = currentItems.map(i => i.id === currentItem.id ? newItem : i);
                } else {
                    newItemsList = [newItem, ...currentItems];
                }

                window.Store.setState({ items: newItemsList });
                close();
            }
        });
    };

    return { open, close, init };
})();
