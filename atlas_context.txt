window.Router = (() => {
    // Чтение URL (Deep Linking)
    const parseUrl = () => {
        const hash = window.location.hash.slice(1) || '/';
        const [path, queryString] = hash.split('?');
        
        const params = {};
        if (queryString) {
            const urlParams = new URLSearchParams(queryString);
            for (const [key, value] of urlParams) {
                params[key] = value;
            }
        }
        return { path: path || '/', params };
    };

    // Запись параметров в URL без мусора в истории (Replace) или с ним
    const setQueryParams = (newParams, replace = false) => {
        const current = parseUrl();
        const search = new URLSearchParams(current.params);
        
        for (const [key, val] of Object.entries(newParams)) {
            if (val === null || val === undefined || val === '') {
                search.delete(key);
            } else {
                search.set(key, val);
            }
        }
        
        const queryStr = search.toString();
        const newHash = queryStr ? `${current.path}?${queryStr}` : current.path;
        
        if (replace) {
            window.history.replaceState(null, '', `#${newHash}`);
            handleRouteChange(); // Триггерим вручную, т.к. replaceState не вызывает hashchange
        } else {
            window.location.hash = newHash;
        }
    };

    const handleRouteChange = () => {
        const route = parseUrl();
        console.log(`[Router] Навигация: ${route.path}`, route.params);
        
        const updates = { 
            route, 
            searchQuery: route.params.q ? route.params.q.toLowerCase() : '' 
        };

        // Запоминаем контекст каталога для Умной кнопки "Назад"
        // Игнорируем URL с открытыми окнами (modal), чтобы не вернуться в форму
        if (route.path === '/' || route.path === '/tierlist') {
            if (!route.params.modal) {
                updates.lastCatalogRoute = window.location.hash || '#/';
            }
        }

        window.Store.setState(updates);
    };

    const init = () => {
        window.addEventListener('hashchange', handleRouteChange);
        handleRouteChange(); // Первый рендер
    };

    return { init, parseUrl, setQueryParams };
})();

window.Store = (() => {
    let state = {
        items: [],
        isAdmin: false,
        route: { path: '/', params: {} },
        searchQuery: '',
        lastCatalogRoute: '#/' // Храним последнюю категорию/поиск для кнопки возврата
    };

    const listeners = [];

    const init = () => {
        try {
            const localData = localStorage.getItem('atlas_data');
            if (localData) {
                state.items = JSON.parse(localData).items;
                console.log("[Store] Данные загружены из LocalStorage");
            } else if (window.__ATLAS_INITIAL_DATA__) {
                state.items = window.__ATLAS_INITIAL_DATA__.items;
                console.log("[Store] Данные загружены из манифеста");
            }
        } catch (error) {
            console.error("[Store] Ошибка загрузки данных:", error);
            state.items = [];
        }
    };

    const getState = () => state;

    const setState = (updates) => {
        state = { ...state, ...updates };
        
        if (updates.items) {
            localStorage.setItem('atlas_data', JSON.stringify({ items: state.items }));
        }

        listeners.forEach(listener => listener(state));
    };

    const subscribe = (listener) => {
        listeners.push(listener);
        listener(state); // Вызов для initial render
    };

    const exportData = () => {
        const dataStr = `window.__ATLAS_INITIAL_DATA__ = ${JSON.stringify({
            metadata: { lastUpdated: new Date().toISOString(), version: "1.5" },
            items: state.items
        }, null, 4)};`;
        
        const blob = new Blob([dataStr], { type: "text/javascript" });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = 'catalog.js';
        a.click();
        URL.revokeObjectURL(url);
    };

    return { init, getState, setState, subscribe, exportData };
})();

window.Helpers = {
    // Защита от XSS-инъекций
    escapeHtml: (unsafe) => {
        if (!unsafe) return '';
        return unsafe.toString().replace(/[&<"'>]/g, (match) => {
            const dict = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' };
            return dict[match];
        });
    },

    // Текстовые статусы для Модалки (Чекбоксы)
    getStatusDict: (type) => {
        if (type === 'games') {
            return {
                'played_self': 'Играл(а) сам(а)',
                'watched_youtube': 'Смотрел(а) на YouTube',
                'watched_reaction': 'Смотрел(а) реакцию',
                'completed': 'Пройдена',
                'in_progress': 'В процессе',
                'dropped': 'Брошена',
                'planned': 'В планах'
            };
        }
        if (type === 'literature') {
            return {
                'played_self': 'Читал(а) сам(а)',
                'watched_youtube': 'Слушал(а) аудиокнигу',
                'watched_reaction': 'Читал(а) с кем-то',
                'completed': 'Прочитана',
                'in_progress': 'В процессе',
                'dropped': 'Брошена',
                'planned': 'В планах'
            };
        }
        return {
            'played_self': 'Смотрел(а) сам(а)',
            'watched_youtube': 'Смотрел(а) на YouTube',
            'watched_reaction': 'Смотрел(а) реакцию',
            'completed': 'Просмотрено',
            'in_progress': 'В процессе',
            'dropped': 'Брошено',
            'planned': 'В планах'
        };
    },

    // Визуальные иконки статусов для Карточки
    getStatusVisuals: (status, type) => {
        // Базовые тексты (если тип не передан)
        let labels = window.Helpers.getStatusDict(type);
        
        switch (status) {
            // Взаимодействие
            case 'played_self':
                let icon = window.Icons.tv;
                if (type === 'games') icon = window.Icons.gamepad;
                if (type === 'literature') icon = window.Icons.book;
                return { svg: icon, colorClass: 'color-primary', title: labels['played_self'] };
            
            case 'watched_youtube':
                return { svg: window.Icons.youtube, colorClass: 'color-youtube', title: labels['watched_youtube'] };
            
            case 'watched_reaction':
                return { svg: window.Icons.users, colorClass: 'color-primary', title: labels['watched_reaction'] };
            
            // Прогресс
            case 'completed':
                return { svg: window.Icons.checkCircle, colorClass: 'color-success', title: labels['completed'] };
            
            case 'in_progress':
                return { svg: window.Icons.clock, colorClass: 'color-info', title: labels['in_progress'] };
            
            case 'dropped':
                return { svg: window.Icons.xCircle, colorClass: 'color-danger', title: labels['dropped'] };
            
            case 'planned':
                return { svg: window.Icons.bookmark, colorClass: 'color-muted', title: labels['planned'] };
            
            default:
                return null;
        }
    }
};

window.CatalogView = (() => {
    const render = (state) => {
        const { items, route, searchQuery, isAdmin } = state;
        const currentType = route.params.type || 'all';
        
        let filteredItems = items.filter(item => {
            const matchType = currentType === 'all' || item.type === currentType;
            const matchSearch = item.title.toLowerCase().includes(searchQuery);
            return matchType && matchSearch;
        });

        if (filteredItems.length === 0) {
            return `
                <div class="empty-state">
                    <div class="empty-state-icon">${window.Icons.emptyBox}</div>
                    <h2>Ничего не найдено</h2>
                    <p>Здесь пока пусто. Попробуйте изменить фильтры или добавьте новую запись.</p>
                </div>
            `;
        }

        const grouped = {};
        filteredItems.forEach(item => {
            const colls = (item.collections && item.collections.length > 0) ? item.collections : ['Без категории'];
            colls.forEach(c => {
                if (!grouped[c]) grouped[c] = [];
                grouped[c].push(item);
            });
        });

        const sortedCategories = Object.keys(grouped).sort((a, b) => {
            if (a === 'Без категории') return 1;
            if (b === 'Без категории') return -1;
            return a.localeCompare(b);
        });

        return sortedCategories.map(category => `
            <section class="collection-group">
                <h2 class="collection-title">
                    ${window.Helpers.escapeHtml(category)} <span>${grouped[category].length}</span>
                </h2>
                <div class="grid">
                    ${grouped[category].map(item => window.CardComponent.render(item, isAdmin)).join('')}
                </div>
            </section>
        `).join('');
    };

    const init = () => {
        const main = document.getElementById('app-main');
        
        main.addEventListener('click', (e) => {
            // Обработка кнопки удаления
            const deleteBtn = e.target.closest('.delete-btn');
            if (deleteBtn) {
                e.preventDefault(); // Блокируем переход по ссылке карточки
                
                const id = deleteBtn.dataset.id;
                if (confirm('Вы уверены, что хотите удалить эту запись?')) {
                    const newItems = window.Store.getState().items.filter(i => i.id !== id);
                    window.Store.setState({ items: newItems });
                    if (window.location.hash.includes(`id=${id}`)) {
                        window.location.hash = '#/';
                    }
                }
                return;
            }

            // Обработка кнопки редактирования
            const editBtn = e.target.closest('.edit-btn');
            if (editBtn) {
                e.preventDefault(); // Блокируем переход по ссылке карточки
                
                const id = editBtn.dataset.id;
                window.Router.setQueryParams({ modal: 'edit', id: id });
                return;
            }
        });
    };

    return { render, init };
})();

window.IconsView = (() => {
    // Словарь категорий: ключи - названия категорий, значения - массивы имен иконок
    const categoryMap = {
        'Навигация и Хедер': ['search', 'lock', 'unlock', 'plus', 'download'],
        'Управление и UI': ['edit', 'trash', 'close'],
        'Типы медиа': ['gamepad', 'book', 'tv', 'youtube', 'users'],
        'Статусы прохождения': ['checkCircle', 'xCircle', 'clock', 'bookmark'],
        'Мета-данные (Карточки)': ['chart', 'calendar', 'document', 'star'],
        'Иллюстрации': ['emptyBox']
    };

    const renderCategory = (title, keys) => {
        const gridHtml = keys.map(key => `
            <div class="icon-card" data-key="${key}" title="Нажми, чтобы скопировать window.Icons.${key}">
                <div class="icon-svg-wrapper">
                    ${window.Icons[key]}
                </div>
                <div class="icon-name">${key}</div>
            </div>
        `).join('');

        return `
            <section class="icons-category">
                <h2 class="icons-category-title">
                    ${title} <span>${keys.length}</span>
                </h2>
                <div class="icons-grid">
                    ${gridHtml}
                </div>
            </section>
        `;
    };

    const render = (state) => {
        if (!state.isAdmin) {
            return `
                <div class="empty-state">
                    <div class="empty-state-icon">${window.Icons.lock || '🔒'}</div>
                    <h2>Доступ запрещен</h2>
                    <p>Эта страница доступна только администратору.</p>
                </div>
            `;
        }

        const allKeys = Object.keys(window.Icons);
        const usedKeys = new Set();
        let contentHtml = '';

        // 1. Рендерим известные категории
        for (const [catName, keys] of Object.entries(categoryMap)) {
            // Берем только те иконки, которые реально существуют в window.Icons
            const availableKeys = keys.filter(k => window.Icons[k]);
            
            if (availableKeys.length > 0) {
                availableKeys.forEach(k => usedKeys.add(k));
                contentHtml += renderCategory(catName, availableKeys);
            }
        }

        // 2. Ищем иконки, которые не попали ни в одну категорию
        const uncategorizedKeys = allKeys.filter(k => !usedKeys.has(k));
        
        // 3. Рендерим "Остальные", если есть новые иконки из скрипта
        if (uncategorizedKeys.length > 0) {
            contentHtml += renderCategory('Новые / Без категории', uncategorizedKeys);
        }

        return `
            <div class="admin-page-header">
                <h1>Библиотека иконок</h1>
                <p>Всего загружено: <strong>${allKeys.length}</strong>. Кликните на карточку, чтобы скопировать вызов JS-переменной.</p>
            </div>
            
            <div id="icons-gallery">
                ${contentHtml}
            </div>
            
            <div id="copy-toast" class="toast">Скопировано!</div>
        `;
    };

    const init = () => {
        const main = document.getElementById('app-main');
        
        main.addEventListener('click', (e) => {
            const card = e.target.closest('.icon-card');
            if (card) {
                const key = card.dataset.key;
                const copyText = `window.Icons.${key}`;
                
                navigator.clipboard.writeText(copyText).then(() => {
                    const toast = document.getElementById('copy-toast');
                    if (toast) {
                        toast.innerText = `Скопировано: ${copyText}`;
                        toast.classList.add('show');
                        setTimeout(() => {
                            toast.classList.remove('show');
                        }, 2000);
                    }
                });
            }
        });
    };

    return { render, init };
})();

window.TierlistView = (() => {
    const tiers = ['S', 'A', 'B', 'C', 'D'];

    const render = (state) => {
        const { items, route, searchQuery } = state;
        const filterType = route.params.filter || 'all'; // Фильтр внутри тир-листа
        
        let targetItems = items.filter(i => i.tier && tiers.includes(i.tier));

        if (filterType !== 'all') {
            targetItems = targetItems.filter(i => i.type === filterType);
        }
        if (searchQuery) {
            targetItems = targetItems.filter(i => i.title.toLowerCase().includes(searchQuery));
        }

        const grouped = { S: [], A: [], B: [], C: [], D: [] };
        targetItems.forEach(item => grouped[item.tier].push(item));

        return `
            <div style="margin-bottom: var(--spacing-lg); display: flex; gap: 10px;">
                <a href="#/tierlist?filter=all" class="btn ${filterType === 'all' ? 'btn-primary' : 'btn-secondary'}">Все</a>
                <a href="#/tierlist?filter=games" class="btn ${filterType === 'games' ? 'btn-primary' : 'btn-secondary'}">Игры</a>
                <a href="#/tierlist?filter=series" class="btn ${filterType === 'series' ? 'btn-primary' : 'btn-secondary'}">Аниме/Сериалы</a>
            </div>

            <div class="tier-board">
                ${tiers.map(tier => `
                    <div class="tier-row">
                        <div class="tier-label tier-${tier}">${tier}</div>
                        <div class="tier-items">
                            ${grouped[tier].map(item => `
                                <div class="tier-item-card">
                                    <img src="${window.Helpers.escapeHtml(item.poster)}" alt="${window.Helpers.escapeHtml(item.title)}" onerror="this.style.display='none'">
                                    <div class="tier-item-tooltip">${window.Helpers.escapeHtml(item.title)}</div>
                                </div>
                            `).join('')}
                        </div>
                    </div>
                `).join('')}
            </div>
        `;
    };

    return { render };
})();
